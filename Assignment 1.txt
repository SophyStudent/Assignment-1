let nextId = 1;

function generateId() {
    const currentId = nextId; // 1. Save current number (1)
    nextId = nextId + 1;      // 2. Increment counter for next time (2)
    return currentId;         // 3. Return saved number (1)
}



nextId++ means:
Use the current value first, then increase it.
So when:
nextId = 1
and we do:
return nextId++;
JavaScript does:
1. Return 1
2. Then change nextId to 2
So the first ID is 1, not 2.
If it were:
return ++nextId;
then it would increase first:
1. Change 1 → 2
2. Return 2
That's the difference between post-increment ++ and pre-increment ++.
